### T04 LLM 适配器

**做什么**：实现统一的 LLM 调用层

**文件**：`src/llm/adapter.py`

```python
import json
import os
import litellm
from pathlib import Path


class LLMAdapter:
    def __init__(self, config: dict):
        self.provider = config.get("provider", "deepseek")
        self.model = config.get("model", "deepseek-chat")
        self.api_key = config.get("api_key", "") or os.getenv("LLM_API_KEY", "")
        self.base_url = config.get("base_url", "")
        self.temperature = config.get("temperature", 0.3)
        self.max_tokens = config.get("max_tokens", 4096)
        self.timeout = config.get("timeout", 120)
        self.max_retries = config.get("max_retries", 3)

    async def generate(
        self,
        user_prompt: str,
        system_prompt: str = "",
        temperature: float | None = None,
        max_tokens: int | None = None,
    ) -> str:
        messages = []
        if system_prompt:
            messages.append({"role": "system", "content": system_prompt})
        messages.append({"role": "user", "content": user_prompt})

        response = await litellm.acompletion(
            model=self.model,
            messages=messages,
            api_key=self.api_key,
            api_base=self.base_url,
            temperature=temperature or self.temperature,
            max_tokens=max_tokens or self.max_tokens,
            timeout=self.timeout,
        )
        return response.choices[0].message.content

    async def generate_json(
        self,
        user_prompt: str,
        system_prompt: str = "",
    ) -> dict:
        raw = await self.generate(user_prompt, system_prompt)
        return self._parse_json(raw)

    def _parse_json(self, text: str) -> dict:
        text = text.strip()
        if text.startswith("```"):
            lines = text.split("\n")
            lines = [l for l in lines if not l.startswith("```")]
            text = "\n".join(lines)
        try:
            return json.loads(text)
        except json.JSONDecodeError:
            start = text.find("{")
            end = text.rfind("}") + 1
            if start >= 0 and end > start:
                return json.loads(text[start:end])
            raise ValueError(f"无法解析JSON: {text[:200]}")
```

import yaml
from pathlib import Path
from jinja2 import Template


class PromptLoader:
    def __init__(self, prompts_dir: str = ""):
        self.prompts_dir = Path(prompts_dir) if prompts_dir else Path(__file__).parent.parent.parent / "config" / "prompts"

    def load(self, name: str) -> dict:
        path = self.prompts_dir / f"{name}.yaml"
        if not path.exists():
            raise FileNotFoundError(f"Prompt模板不存在: {path}")
        with open(path, "r", encoding="utf-8") as f:
            return yaml.safe_load(f)

    def render(self, name: str, variables: dict) -> tuple[str, str]:
        template = self.load(name)
        system_prompt = Template(template.get("system_prompt", "")).render(**variables)
        user_prompt = Template(template.get("user_prompt", "")).render(**variables)
        return system_prompt, user_prompt
```

### T05 存储层

**做什么**：实现项目数据的读写

**文件**：`src/storage/project_store.py`

```python
import json
from pathlib import Path
from src.models.project import Project
from src.models.world_bible import WorldBible
from src.models.entity import Entity
from src.models.chapter import Chapter
from src.models.prompt import Prompt


class ProjectStore:
    def __init__(self, output_dir: str):
        self.base_dir = Path(output_dir)

    def create_project(self, project: Project) -> Path:
        proj_dir = self.base_dir / project.id
        proj_dir.mkdir(parents=True, exist_ok=True)
        (proj_dir / "entities").mkdir(exist_ok=True)
        (proj_dir / "prompts").mkdir(exist_ok=True)
        (proj_dir / "chapters").mkdir(exist_ok=True)
        (proj_dir / "face_anchors").mkdir(exist_ok=True)
        (proj_dir / "characters").mkdir(exist_ok=True)
        (proj_dir / "scenes").mkdir(exist_ok=True)
        (proj_dir / "illustrations").mkdir(exist_ok=True)
        self._save_json(proj_dir / "meta.json", project.model_dump())
        return proj_dir

    def save_world_bible(self, project_id: str, wb: WorldBible):
        path = self.base_dir / project_id / "world_bible.json"
        self._save_json(path, wb.model_dump())

    def load_world_bible(self, project_id: str) -> WorldBible:
        path = self.base_dir / project_id / "world_bible.json"
        data = self._load_json(path)
        return WorldBible(**data)

    def save_chapter(self, project_id: str, chapter: Chapter):
        path = self.base_dir / project_id / "chapters" / f"ch_{chapter.number:04d}.json"
        self._save_json(path, chapter.model_dump())

    def save_entity(self, project_id: str, entity: Entity):
        path = self.base_dir / project_id / "entities" / f"{entity.type}_{entity.id}.json"
        self._save_json(path, entity.model_dump())

    def save_prompt(self, project_id: str, prompt: Prompt):
        path = self.base_dir / project_id / "prompts" / f"{prompt.type}_{prompt.id}.json"
        self._save_json(path, prompt.model_dump())

    def save_prompts_md(self, project_id: str, prompts: list[Prompt], filename: str):
        path = self.base_dir / project_id / "prompts" / filename
        md_content = self._prompts_to_md(prompts)
        path.write_text(md_content, encoding="utf-8")

    def _prompts_to_md(self, prompts: list[Prompt]) -> str:
        lines = []
        for p in prompts:
            lines.append(f"## {p.type} - {p.id}")
            lines.append("")
            if p.face_block_chinese:
                lines.append("### 面容独立块")
                lines.append(f"**中文**: {p.face_block_chinese}")
                lines.append(f"**英文**: {p.face_block_english}")
                lines.append("")
            lines.append("### 中文提示词")
            lines.append(p.chinese_prompt)
            lines.append("")
            lines.append("### 英文提示词")
            lines.append(p.english_prompt)
            lines.append("")
            if p.negative_prompt:
                lines.append("### 反向提示词")
                lines.append(p.negative_prompt)
                lines.append("")
            lines.append("---")
            lines.append("")
        return "\n".join(lines)

    def _save_json(self, path: Path, data: dict):
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    def _load_json(self, path: Path) -> dict:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
```

### T11 实现章节识别

**做什么**：实现正则+LLM兜底的章节识别

**文件**：`src/core/preprocessor.py`

```python
import re
import chardet
from pathlib import Path
from src.models.chapter import Chapter
import uuid


CHAPTER_PATTERNS = [
    re.compile(r"^第[一二三四五六七八九十百千万零\d]+[章节回卷集].*$", re.MULTILINE),
    re.compile(r"^Chapter\s+\d+.*$", re.MULTILINE | re.IGNORECASE),
    re.compile(r"^第\d+章.*$", re.MULTILINE),
    re.compile(r"^\d{1,4}[\.、]\s.*$", re.MULTILINE),
    re.compile(r"^【.*】$", re.MULTILINE),
]


class Preprocessor:
    def __init__(self, config: dict = None):
        self.config = config or {}

    def read_file(self, file_path: str) -> str:
        path = Path(file_path)
        raw = path.read_bytes()
        detected = chardet.detect(raw)
        encoding = detected.get("encoding", "utf-8") or "utf-8"
        try:
            return raw.decode(encoding)
        except (UnicodeDecodeError, LookupError):
            return raw.decode("utf-8", errors="replace")

    def split_chapters(self, text: str, project_id: str = "") -> list[Chapter]:
        for pattern in CHAPTER_PATTERNS:
            matches = list(pattern.finditer(text))
            if len(matches) >= 2:
                return self._split_by_matches(text, matches, project_id)

        return self._split_by_window(text, project_id)

    def _split_by_matches(self, text: str, matches: list, project_id: str) -> list[Chapter]:
        chapters = []
        for i, match in enumerate(matches):
            start = match.start()
            end = matches[i + 1].start() if i + 1 < len(matches) else len(text)
            title = match.group().strip()
            body = text[start:end].replace(title, "", 1).strip()
            chapters.append(Chapter(
                id=str(uuid.uuid4())[:8],
                project_id=project_id,
                number=i + 1,
                title=title,
                text=body,
            ))
        return chapters

    def _split_by_window(self, text: str, project_id: str) -> list[Chapter]:
        window_size = self.config.get("sliding_window_size", 5000)
        chapters = []
        for i in range(0, len(text), window_size):
            chunk = text[i:i + window_size]
            chapters.append(Chapter(
                id=str(uuid.uuid4())[:8],
                project_id=project_id,
                number=len(chapters) + 1,
                title=f"段落{len(chapters) + 1}",
                text=chunk,
            ))
        return chapters
```

### T16 实现 EntityMerger

**做什么**：实现实体消歧与归一

**文件**：`src/core/entity_merger.py`

```python
import uuid
from difflib import SequenceMatcher
from src.models.entity import Entity, EntityType


class EntityMerger:
    def __init__(self, config: dict = None):
        self.config = config or {}
        self.similarity_threshold = self.config.get("dedup_similarity_threshold", 0.85)

    def merge(
        self,
        candidates: list[dict],
        existing: list[Entity],
        project_id: str = "",
    ) -> list[Entity]:
        result = list(existing)
        for cand in candidates:
            matched = self._find_match(cand, result)
            if matched:
                self._merge_into(matched, cand)
            else:
                entity = self._to_entity(cand, project_id)
                result.append(entity)
        return result

    def _find_match(self, candidate: dict, entities: list[Entity]) -> Entity | None:
        cand_name = candidate.get("name", "")
        cand_aliases = candidate.get("aliases", [])
        cand_type = candidate.get("type", "")

        for entity in entities:
            if entity.type.value != cand_type:
                continue
            all_names = [entity.name] + entity.aliases
            all_cand_names = [cand_name] + cand_aliases
            for n1 in all_names:
                for n2 in all_cand_names:
                    if n1 == n2:
                        return entity
                    if self._similarity(n1, n2) >= self.similarity_threshold:
                        return entity
        return None

    def _merge_into(self, entity: Entity, candidate: dict):
        new_aliases = [a for a in candidate.get("aliases", []) if a not in entity.aliases and a != entity.name]
        entity.aliases.extend(new_aliases)
        if candidate.get("source_quote") and candidate["source_quote"] not in [sq.text for sq in entity.source_quotes]:
            from src.models.entity import SourceQuote
            entity.source_quotes.append(SourceQuote(
                chapter=candidate.get("chapter", 0),
                text=candidate["source_quote"],
                location=candidate.get("location", ""),
            ))

    def _to_entity(self, candidate: dict, project_id: str) -> Entity:
        return Entity(
            id=f"{candidate.get('type', 'unknown')}_{str(uuid.uuid4())[:8]}",
            project_id=project_id,
            name=candidate.get("name", ""),
            aliases=candidate.get("aliases", []),
            type=EntityType(candidate.get("type", "character")),
        )

    def _similarity(self, a: str, b: str) -> float:
        return SequenceMatcher(None, a, b).ratio()
```

            return await self._generate_scene(entity, world_bible)
        elif entity.type == EntityType.ITEM:
            return await self._generate_item(entity, world_bible)

    async def _generate_character(self, entity: Entity, wb: WorldBible) -> Prompt:
        system, user = self.prompt_loader.render("character_prompt", {
            "entity_json": json.dumps(entity.model_dump(), ensure_ascii=False, indent=2),
            "world_bible_visual_anchoring": json.dumps(wb.visual_anchoring.model_dump(), ensure_ascii=False, indent=2),
            "source_quotes": self._format_quotes(entity.source_quotes),
        })
        result = await self.llm.generate_json(user, system)
        return self._to_prompt(result, entity, "character", "3:4")

    async def _generate_scene(self, entity: Entity, wb: WorldBible) -> Prompt:
        system, user = self.prompt_loader.render("scene_prompt", {
            "entity_json": json.dumps(entity.model_dump(), ensure_ascii=False, indent=2),
            "world_bible_visual_anchoring": json.dumps(wb.visual_anchoring.model_dump(), ensure_ascii=False, indent=2),
            "world_bible_scene_rules": json.dumps(wb.scene_visual_rules.model_dump(), ensure_ascii=False),
            "source_quotes": self._format_quotes(entity.source_quotes),
        })
        result = await self.llm.generate_json(user, system)
        return self._to_prompt(result, entity, "scene", "16:9")

    async def _generate_item(self, entity: Entity, wb: WorldBible) -> Prompt:
        system, user = self.prompt_loader.render("item_prompt", {
            "entity_json": json.dumps(entity.model_dump(), ensure_ascii=False, indent=2),
            "world_bible_visual_anchoring": json.dumps(wb.visual_anchoring.model_dump(), ensure_ascii=False, indent=2),
            "world_bible_item_rules": json.dumps(wb.item_visual_rules.model_dump(), ensure_ascii=False),
            "source_quotes": self._format_quotes(entity.source_quotes),
        })
        result = await self.llm.generate_json(user, system)
        return self._to_prompt(result, entity, "item", "1:1")

    def _to_prompt(self, result: dict, entity: Entity, prompt_type: str, default_ar: str) -> Prompt:
        params = result.get("parameters", {})
        return Prompt(
            id=str(uuid.uuid4())[:8],
            entity_id=entity.id,
            type=prompt_type,
            world_prefix_chinese=result.get("world_prefix_chinese", ""),
            world_prefix_english=result.get("world_prefix_english", ""),
            face_block_chinese=result.get("face_block_chinese", ""),
            face_block_english=result.get("face_block_english", ""),
            chinese_prompt=result.get("chinese_prompt", ""),
            english_prompt=result.get("english_prompt", ""),
            negative_prompt=result.get("negative_prompt", ""),
            style_tags=result.get("style_tags", []),
            parameters=PromptParameters(
                aspect_ratio=params.get("aspect_ratio", default_ar),
                steps=params.get("steps", 30),
                cfg_scale=params.get("cfg_scale", 7.0),
            ),
        )

    def _format_quotes(self, quotes) -> str:
        if not quotes:
            return "（无原文引用）"
        return "\n".join([f"- [{q.location}] {q.text}" for q in quotes])
```

import asyncio
from pathlib import Path
from src.llm.adapter import LLMAdapter
from src.llm.prompt_loader import PromptLoader
from src.storage.project_store import ProjectStore
from src.core.world_bible_builder import WorldBibleBuilder

class Pipeline:
    def __init__(self, config: dict):
        self.config = config
        llm_config = config.get("llm", {})
        self.llm = LLMAdapter(llm_config)
        self.prompt_loader = PromptLoader(
            str(Path(__file__).parent.parent.parent / "config" / "prompts")
        )
        self.store = ProjectStore(config.get("output", {}).get("dir", "./output"))
        self.wb_builder = WorldBibleBuilder(self.llm, self.prompt_loader)
        self.preprocessor = Preprocessor(config.get("extraction", {}))
        self.extractor = EntityExtractor(self.llm, self.prompt_loader, config.get("extraction", {}))
        self.merger = EntityMerger(config.get("extraction", {}))
        self.attr_builder = AttributeBuilder(self.llm, self.prompt_loader)
        self.prompt_gen = PromptGenerator(self.llm, self.prompt_loader, config.get("prompt", {}))

    async def run(self, input_path: str, output_dir: str = "", stages: list[str] = None) -> Project:
        import uuid
        project = Project(
            id=str(uuid.uuid4())[:8],
            novel_title=Path(input_path).stem,
            input_path=input_path,
        )
        proj_dir = self.store.create_project(project)

        stages = stages or ["world_bible", "extract", "prompt"]

        # Stage 0: WorldBible
        if "world_bible" in stages:
            text = self.preprocessor.read_file(input_path)
            chapters = self.preprocessor.split_chapters(text, project.id)
            analysis_text = "\n".join([ch.text for ch in chapters[:5]])
            wb = await self.wb_builder.analyze(analysis_text, project.novel_title)
            wb.project_id = project.id
            self.store.save_world_bible(project.id, wb)
            project.world_bible_id = wb.id
            project.status = ProjectStatus.WORLD_BIBLE_DONE
        else:
            wb = self.store.load_world_bible(project.id)

        # Stage 1+2: Preprocess + Extract
        entities = []
        if "extract" in stages:
            text = self.preprocessor.read_file(input_path)
            chapters = self.preprocessor.split_chapters(text, project.id)
            for ch in chapters:
                self.store.save_chapter(project.id, ch)

            for ch in chapters:
                candidates = await self.extractor.extract_from_chapter(ch, wb, entities)
                entities = self.merger.merge(candidates, entities, project.id)

            for i, entity in enumerate(entities):
                entities[i] = await self.attr_builder.build_attributes(entity, wb)
                self.store.save_entity(project.id, entities[i])

            project.status = ProjectStatus.EXTRACT_DONE

        # Stage 3: Prompt
        prompts = []
        if "prompt" in stages:
            for entity in entities:
                prompt = await self.prompt_gen.generate(entity, wb)
                prompts.append(prompt)
                self.store.save_prompt(project.id, prompt)

            char_prompts = [p for p in prompts if p.type == "character"]
            scene_prompts = [p for p in prompts if p.type == "scene"]
            item_prompts = [p for p in prompts if p.type == "item"]

            if char_prompts:
                self.store.save_prompts_md(project.id, char_prompts, "characters.md")
            if scene_prompts:
                self.store.save_prompts_md(project.id, scene_prompts, "scenes.md")
            if item_prompts:
                self.store.save_prompts_md(project.id, item_prompts, "items.md")

            project.status = ProjectStatus.PROMPT_DONE

        self.store._save_json(proj_dir / "meta.json", project.model_dump())
        return project
```

### T25 实现 ChatGPT2API 生图后端

**做什么**：基于你的 ChatGPT2API 反代服务，实现生图后端

**核心优势**：
- 无需 GPU，通过本地反代调用 gpt-image-2
- **图片编辑接口**（`/v1/images/edits`）天然支持面部一致性：传入面部锚定图 + 提示词 → 保持面部特征的全身图
- OpenAI SDK 兼容，代码极简

**文件**：`src/render/chatgpt2api_backend.py`

```python
import base64
import io
from pathlib import Path
from openai import OpenAI
from PIL import Image


class ChatGPT2APIBackend:
    def __init__(self, config: dict):
        self.base_url = config.get("base_url", "http://localhost:3000/v1")
        self.api_key = config.get("api_key", "biaooo")
        self.model = config.get("model", "gpt-image-2")
        self.client = OpenAI(api_key=self.api_key, base_url=self.base_url)

    def is_available(self) -> bool:
        try:
            self.client.models.list()
            return True
        except Exception:
            return False

    async def text_to_image(
        self,
        prompt: str,
        size: str = "1024x1024",
        n: int = 1,
    ) -> list[bytes]:
        """
        文生图：POST /v1/images/generations

        Args:
            prompt: 图片生成提示词
            size: 图片尺寸，如 "1024x1024", "1024x1536", "1536x1024"
            n: 生成数量

        Returns:
            图片 bytes 列表
        """
        response = self.client.images.generate(
            model=self.model,
            prompt=prompt,
            n=n,
            size=size,
            response_format="b64_json",
        )
        results = []
        for item in response.data:
            img_bytes = base64.b64decode(item.b64_json)
            results.append(img_bytes)
        return results

    async def edit_image(
        self,
        prompt: str,
        image_bytes: bytes | list[bytes],
        size: str = "1024x1024",
        n: int = 1,
    ) -> list[bytes]:
        """
        图生图（图片编辑）：POST /v1/images/edits
        这是面部一致性的核心接口！

        传入面部锚定图 + 提示词 → 保持面部特征的全身图

        Args:
            prompt: 编辑提示词（描述目标图片）
            image_bytes: 参考图片（面部锚定图），支持单张或多张
            size: 输出尺寸
            n: 生成数量

        Returns:
            图片 bytes 列表
        """
        if isinstance(image_bytes, bytes):
            image_bytes = [image_bytes]

        images = []
        for img_b in image_bytes:
            img_file = io.BytesIO(img_b)
            img_file.name = "reference.png"
            images.append(img_file)

        response = self.client.images.edit(
            model=self.model,
            image=images[0] if len(images) == 1 else images,
            prompt=prompt,
            n=n,
            size=size,
        )

        results = []
        for item in response.data:
            if hasattr(item, "b64_json") and item.b64_json:
                img_bytes = base64.b64decode(item.b64_json)
            elif hasattr(item, "url") and item.url:
                import httpx
                resp = httpx.get(item.url)
                img_bytes = resp.content
            else:
                continue
            results.append(img_bytes)
        return results

    async def generate_face_anchor(
        self,
        face_prompt: str,
        size: str = "1024x1024",
    ) -> bytes:
        """
        生成面部锚定图（正面特写，纯色背景）

        Args:
            face_prompt: 面部锚定图专用提示词

        Returns:
            面部锚定图 bytes
        """
        results = await self.text_to_image(face_prompt, size=size, n=1)
        return results[0]

    async def generate_character_with_face(
        self,
        character_prompt: str,
        face_anchor_bytes: bytes,
        size: str = "1024x1536",
    ) -> bytes:
        """
        基于面部锚定图生成角色全身图（面部锁定）

        流程：
        1. 传入面部锚定图作为参考
        2. 提示词描述角色全身
        3. gpt-image-2 的图片编辑能力会保持面部特征

        Args:
            character_prompt: 角色全身描述提示词
            face_anchor_bytes: 面部锚定图
            size: 输出尺寸（角色图默认竖版）

        Returns:
            角色全身图 bytes
        """
        results = await self.edit_image(
            prompt=character_prompt,
            image_bytes=face_anchor_bytes,
            size=size,
            n=1,
        )
        return results[0]

    async def generate_scene(
        self,
        scene_prompt: str,
        style_reference: bytes | None = None,
        size: str = "1536x1024",
    ) -> bytes:
        """
        生成场景背景图

        Args:
            scene_prompt: 场景描述提示词
            style_reference: 可选的风格参考图（保持画风统一）
            size: 输出尺寸（场景图默认横版）

        Returns:
            场景图 bytes
        """
        if style_reference:
            results = await self.edit_image(
                prompt=scene_prompt,
                image_bytes=style_reference,
                size=size,
                n=1,
            )
        else:
            results = await self.text_to_image(scene_prompt, size=size, n=1)
        return results[0]

    async def generate_item(
        self,
        item_prompt: str,
        size: str = "1024x1024",
    ) -> bytes:
        """
        生成物品设定图

        Args:
            item_prompt: 物品描述提示词
            size: 输出尺寸（物品图默认方形）

        Returns:
            物品图 bytes
        """
        results = await self.text_to_image(item_prompt, size=size, n=1)
        return results[0]
```

### T27 实现角色全身图生成（面部锁定）

**做什么**：基于面部锚定图，通过 ChatGPT2API 的图片编辑接口生成面部一致的全身图

**核心原理**：
```
面部锚定图 (PNG) + 角色全身提示词 → /v1/images/edits → 保持面部的全身图
```

ChatGPT2API 的图片编辑接口会参考输入图片的视觉特征，在生成新图时保持一致性。
这比 ComfyUI 的 IP-Adapter 方案更简单，无需 GPU，无需训练。

**文件**：`src/core/image_generator.py`

```python
from pathlib import Path
from src.models.entity import Entity, EntityType
from src.models.prompt import Prompt
from src.models.world_bible import WorldBible
from src.render.chatgpt2api_backend import ChatGPT2APIBackend


class ImageGenerator:
    def __init__(self, backend: ChatGPT2APIBackend, config: dict = None):
        self.backend = backend
        self.config = config or {}

    async def generate_character(
        self,
        entity: Entity,
        prompt: Prompt,
        face_anchor_path: str,
        output_dir: str,
    ) -> str:
        """
        基于面部锚定图生成角色全身图

        Args:
            entity: 角色实体
            prompt: 角色提示词
            face_anchor_path: 面部锚定图路径
            output_dir: 输出目录

        Returns:
            生成图路径
        """
        face_anchor_bytes = Path(face_anchor_path).read_bytes()
        image_bytes = await self.backend.generate_character_with_face(
            character_prompt=prompt.english_prompt,
            face_anchor_bytes=face_anchor_bytes,
            size="1024x1536",
        )
        output_path = Path(output_dir) / "characters" / f"{entity.id}.png"
        output_path.parent.mkdir(parents=True, exist_ok=True)
        output_path.write_bytes(image_bytes)
        return str(output_path)

    async def generate_scene(
        self,
        entity: Entity,
        prompt: Prompt,
        style_reference_path: str | None = None,
        output_dir: str = "",
    ) -> str:
        """
        生成场景背景图

        如果有风格参考图（如第一张场景图），通过 edit_image 保持画风统一
        """
        style_ref = None
        if style_reference_path and Path(style_reference_path).exists():
            style_ref = Path(style_reference_path).read_bytes()

        image_bytes = await self.backend.generate_scene(
            scene_prompt=prompt.english_prompt,
            style_reference=style_ref,
            size="1536x1024",
        )
        output_path = Path(output_dir) / "scenes" / f"{entity.id}.png"
        output_path.parent.mkdir(parents=True, exist_ok=True)
        output_path.write_bytes(image_bytes)
        return str(output_path)

    async def generate_item(
        self,
        entity: Entity,
        prompt: Prompt,
        output_dir: str = "",
    ) -> str:
        """
        生成物品设定图
        """
        image_bytes = await self.backend.generate_item(
            item_prompt=prompt.english_prompt,
            size="1024x1024",
        )
        output_path = Path(output_dir) / "items" / f"{entity.id}.png"
        output_path.parent.mkdir(parents=True, exist_ok=True)
        output_path.write_bytes(image_bytes)
        return str(output_path)

    async def generate_all(
        self,
        entities: list[Entity],
        prompts: list[Prompt],
        world_bible: WorldBible,
        face_anchor_dir: str,
        output_dir: str,
    ) -> dict[str, list[str]]:
        """
        批量生成所有图片

        Returns:
            {"characters": [path, ...], "scenes": [path, ...], "items": [path, ...]}
        """
        prompt_map = {p.entity_id: p for p in prompts}
        results = {"characters": [], "scenes": [], "items": []}
        first_scene_path = None

        for entity in entities:
            prompt = prompt_map.get(entity.id)
            if not prompt:
                continue

            if entity.type == EntityType.CHARACTER:
                face_anchor_path = str(Path(face_anchor_dir) / f"{entity.id}.png")
                if Path(face_anchor_path).exists():
                    path = await self.generate_character(entity, prompt, face_anchor_path, output_dir)
                    results["characters"].append(path)

            elif entity.type == EntityType.SCENE:
                path = await self.generate_scene(entity, prompt, first_scene_path, output_dir)
                results["scenes"].append(path)
                if not first_scene_path:
                    first_scene_path = path

            elif entity.type == EntityType.ITEM:
                path = await self.generate_item(entity, prompt, output_dir)
                results["items"].append(path)

        return results
```

### T28 实现面部一致性验证（多模态LLM方案）

**做什么**：使用多模态 LLM 验证面部一致性

**方案选择**：由于 gpt-image-2 不像 SD 那样有特征向量可提取，我们采用两种验证方式：
1. **多模态 LLM 验证**（主要）：用 Qwen-VL / GPT-4o 对比两张图的面部相似度
2. **DeepFace 验证**（可选）：用 DeepFace 计算面部 embedding 余弦相似度

**文件**：`src/render/face_consistency.py`

```python
import base64
from src.llm.adapter import LLMAdapter


class FaceConsistencyChecker:
    def __init__(self, llm: LLMAdapter, config: dict = None):
        self.llm = llm
        self.config = config or {}
        self.similarity_threshold = self.config.get("similarity_threshold", 0.7)
        self.max_retries = self.config.get("max_retries", 3)

    async def verify(
        self,
        anchor_image_path: str,
        generated_image_path: str,
        entity_name: str = "",
    ) -> dict:
        """
        使用多模态 LLM 验证面部一致性

        Returns:
            {
                "is_consistent": bool,
                "face_similarity_score": float,
                "details": str,
                "issues": list[str]
            }
        """
        anchor_b64 = self._image_to_b64(anchor_image_path)
        generated_b64 = self._image_to_b64(generated_image_path)

        system_prompt = (
            "你是一位专业的面部特征对比专家。"
            "你需要对比两张图片中人物的面部特征，判断是否为同一人。"
            "请从以下维度逐一对比：脸型、眼型、眉形、鼻型、嘴型、发色、标志性特征。"
            "输出合法的 JSON 格式。"
        )

        user_prompt = f"""请对比以下两张图片中人物的面部特征。

角色名称：{entity_name}

图1（面部锚定图）：data:image/png;base64,{anchor_b64[:100]}...
图2（生成的角色图）：data:image/png;base64,{generated_b64[:100]}...

请严格按以下 JSON 格式输出：

{{
  "is_same_person": true/false,
  "face_similarity_score": 0.0-1.0,
  "face_shape_match": true/false,
  "eye_shape_match": true/false,
  "eyebrow_shape_match": true/false,
  "nose_shape_match": true/false,
  "mouth_shape_match": true/false,
  "hair_color_match": true/false,
  "distinguishing_features_match": true/false,
  "issues": ["问题描述1", "问题描述2"],
  "suggestion": "改进建议"
}}"""

        result = await self.llm.generate_json(user_prompt, system_prompt)

        score = result.get("face_similarity_score", 0.0)
        is_consistent = score >= self.similarity_threshold and result.get("is_same_person", False)

        return {
            "is_consistent": is_consistent,
            "face_similarity_score": score,
            "details": result.get("suggestion", ""),
            "issues": result.get("issues", []),
        }

    async def verify_with_retry(
        self,
        anchor_image_path: str,
        generated_image_path: str,
        entity_name: str = "",
        max_retries: int = None,
    ) -> dict:
        """
        验证 + 自动重试（如果不一致则标记需要重新生成）
        """
        retries = max_retries or self.max_retries
        for attempt in range(retries):
            result = await self.verify(anchor_image_path, generated_image_path, entity_name)
            if result["is_consistent"]:
                result["attempts"] = attempt + 1
                return result
        result["attempts"] = retries
        return result

    def _image_to_b64(self, image_path: str) -> str:
        from pathlib import Path
        img_bytes = Path(image_path).read_bytes()
        return base64.b64encode(img_bytes).decode("utf-8")
```

### 面部一致性方案对比

| 方案 | 后端 | 原理 | 需要GPU | 一致性 | 复杂度 |
|------|------|------|---------|--------|--------|
| **edit_image（推荐）** | ChatGPT2API | 图片编辑接口，参考图+提示词 | ❌ | ★★★★ | 低 |
| IP-Adapter FaceID | ComfyUI | 面部特征注入扩散模型 | ✅ | ★★★★★ | 高 |
| 参考图引导 | DALL-E 3 | 提示词中描述面部 | ❌ | ★★★ | 低 |

### ChatGPT2API 面部一致性工作流

```
Step 1: text_to_image(面部锚定图提示词) → face_anchor.png
         ↓
Step 2: edit_image(角色全身提示词, face_anchor.png) → character_full.png
         ↓
Step 3: verify(face_anchor.png, character_full.png) → 相似度评分
         ↓ 不通过
Step 2: edit_image(调整后提示词, face_anchor.png) → character_full_v2.png
         ↓ 通过
Step 4: 保存最终图片
```

### 多角色场景图生成

```
对于包含多个角色的分镜图：
1. 每个角色先各自生成面部锚定图
2. 使用 edit_image 传入第一个角色的锚定图
3. 在提示词中详细描述所有角色的面部特征
4. gpt-image-2 会尽量保持参考图角色的面部
5. 其他角色通过提示词中的面容独立块描述
```
