# AI拆书生图 - 生图模块使用示例

## 快速开始

### 1. 初始化 ChatGPT2API 后端

```python
import asyncio
from src.render.chatgpt2api_backend import ChatGPT2APIBackend

async def main():
    # 初始化后端
    backend = ChatGPT2APIBackend({
        "base_url": "http://localhost:3000/v1",
        "api_key": "biaooo",
        "model": "gpt-image-2",
    })
    
    # 检查服务是否可用
    if backend.is_available():
        print("ChatGPT2API 服务可用")
    else:
        print("ChatGPT2API 服务不可用")

asyncio.run(main())
```

### 2. 生成面部锚定图

```python
import asyncio
from pathlib import Path
from src.render.chatgpt2api_backend import ChatGPT2APIBackend

async def generate_face_anchor():
    backend = ChatGPT2APIBackend({
        "base_url": "http://localhost:3000/v1",
        "api_key": "biaooo",
        "model": "gpt-image-2",
    })
    
    # 生成面部锚定图提示词
    face_prompt = (
        "Close-up portrait of a beautiful young Chinese woman, "
        "fair skin, delicate features, natural makeup, "
        "long black hair, neutral expression, "
        "pure white background, centered composition, "
        "high detail, realistic photography style"
    )
    
    # 生成图片
    image_bytes = await backend.generate_face_anchor(face_prompt)
    
    # 保存图片
    output_path = Path("output/face_anchors/char_001.png")
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_bytes(image_bytes)
    
    print(f"面部锚定图已保存: {output_path}")
    return str(output_path)

asyncio.run(generate_face_anchor())
```

### 3. 基于面部锚定图生成角色全身图

```python
import asyncio
from pathlib import Path
from src.render.chatgpt2api_backend import ChatGPT2APIBackend

async def generate_character():
    backend = ChatGPT2APIBackend({
        "base_url": "http://localhost:3000/v1",
        "api_key": "biaooo",
        "model": "gpt-image-2",
    })
    
    # 读取面部锚定图
    face_anchor_path = Path("output/face_anchors/char_001.png")
    face_anchor_bytes = face_anchor_path.read_bytes()
    
    # 角色全身图提示词
    character_prompt = (
        "Full body portrait of the same woman from the reference image, "
        "wearing elegant traditional Chinese Hanfu in white and silver colors, "
        "standing in a moonlit courtyard with cherry blossoms falling, "
        "medieval Chinese fantasy style, cinematic lighting, "
        "detailed fabric textures, high quality illustration"
    )
    
    # 生成角色全身图
    image_bytes = await backend.generate_character_with_face(
        character_prompt=character_prompt,
        face_anchor_bytes=face_anchor_bytes,
        size="1024x1536"
    )
    
    # 保存图片
    output_path = Path("output/characters/char_001.png")
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_bytes(image_bytes)
    
    print(f"角色全身图已保存: {output_path}")

asyncio.run(generate_character())
```

### 5. 使用面部一致性检查器

```python
import asyncio
from pathlib import Path
from src.render.face_consistency import FaceConsistencyChecker
from src.llm.adapter import LLMAdapter

async def verify_face_consistency():
    # 初始化LLM适配器
    llm = LLMAdapter({
        "provider": "deepseek",
        "model": "deepseek-chat",
        "api_key": "your-api-key",
    })
    
    # 初始化面部一致性检查器
    checker = FaceConsistencyChecker(llm, {
        "similarity_threshold": 0.7,
        "max_retries": 3,
    })
    
    # 验证两张图的面部一致性
    result = await checker.verify(
        anchor_image_path="output/face_anchors/char_001.png",
        generated_image_path="output/characters/char_001.png",
        entity_name="林婉儿"
    )
    
    print(f"面部一致性验证结果:")
    print(f"  - 是否一致: {result['is_consistent']}")
    print(f"  - 相似度分数: {result['face_similarity_score']}")
    print(f"  - 问题列表: {result['issues']}")
    print(f"  - 详细建议: {result['details']}")

asyncio.run(verify_face_consistency())
```

### 6. 批量生成所有图片

```python
import asyncio
from pathlib import Path
from src.render.chatgpt2api_backend import ChatGPT2APIBackend
from src.core.image_generator import ImageGenerator
from src.models.entity import Entity, EntityType
from src.models.prompt import Prompt

async def batch_generate():
    # 初始化
    backend = ChatGPT2APIBackend({
        "base_url": "http://localhost:3000/v1",
        "api_key": "biaooo",
        "model": "gpt-image-2",
    })
    
    generator = ImageGenerator(backend, {
        "default_sizes": {
            "character": "1024x1536",
            "scene": "1536x1024",
            "item": "1024x1024",
            "face_anchor": "1024x1024",
        }
    })
    
    # 准备实体和提示词
    entities = [
        Entity(id="char_001", name="林婉儿", type=EntityType.CHARACTER),
        Entity(id="scene_001", name="云隐峰", type=EntityType.SCENE),
        Entity(id="item_001", name="玄冰剑", type=EntityType.ITEM),
    ]
    
    prompts = [
        Prompt(id="p_001", entity_id="char_001", type="character", 
               english_prompt="A beautiful Chinese cultivator..."),
        Prompt(id="p_002", entity_id="scene_001", type="scene",
               english_prompt="Ancient mountain peak with clouds..."),
        Prompt(id="p_003", entity_id="item_001", type="item",
               english_prompt="Elegant Chinese sword with ice patterns..."),
    ]
    
    # 批量生成
    results = await generator.generate_all(
        entities=entities,
        prompts=prompts,
        world_bible=None,
        face_anchor_dir="output/face_anchors",
        output_dir="output"
    )
    
    print("批量生成结果:")
    print(f"  - 角色图: {len(results['characters'])} 张")
    print(f"  - 场景图: {len(results['scenes'])} 张")
    print(f"  - 物品图: {len(results['items'])} 张")

asyncio.run(batch_generate())
```

## 注意事项

1. **API 服务**: 确保 ChatGPT2API 反代服务在 `http://localhost:3000` 运行
2. **API Key**: 使用配置中的 API Key 进行认证
3. **模型选择**: 推荐使用 `gpt-image-2` 模型
4. **异步调用**: 所有生图方法都是异步的，需要使用 `await` 调用
5. **错误处理**: API 调用可能失败，建议添加适当的错误处理

## 面部一致性工作流

```
面部锚定图生成 (text_to_image)
         ↓
面部锚定图保存 (face_anchors/{id}.png)
         ↓
角色全身图生成 (edit_image + face_anchor)
         ↓
面部一致性验证 (FaceConsistencyChecker)
         ↓ 不通过
重新生成 (使用调整后的提示词)
         ↓ 通过
保存最终图片
```
